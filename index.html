<script>

function displayMedia() {

const input = document.getElementById('insta-link').value.trim();

const preview = document.getElementById('preview-area');

if (!input) { alert('لطفاً لینک پست یا ریلز اینستاگرام را وارد کنید.'); return; }

preview.innerHTML = '<p style="color: var(--text-secondary);">در حال دریافت...</p>';

let cleanUrl = input.split('?')[0];

if (!cleanUrl.endsWith('/')) cleanUrl += '/';

cleanUrl += 'embed';

setTimeout(() => {

preview.innerHTML = `<iframe src="${cleanUrl}" width="400" height="480" frameborder="0" scrolling="no" allowtransparency="true"></iframe>`;

}, 500);

}

</script>
